Мы получили ошибку "list.map(...).join is not a function" — это значит, что в одном из NPC поле списка (например, языки или сопротивления) пришло не массивом, а строкой или объектом. Мы добавим проверку типа и безопасную обработку.

Исправленный макрос (замените только функцию translateList и вызовы):

javascript
function translateList(list, dict) {
    if (!list) return '';
    // Если строка, разбиваем по запятой
    if (typeof list === 'string') {
        list = list.split(',').map(s => s.trim());
    }
    if (!Array.isArray(list)) return '';
    return list.map(item => dict[item.toLowerCase()] || item).join(', ');
}

можешь написать полный итоговый не могу найти где заменить
Мы почти у цели. Действительно, проще дать полный исправленный макрос, который можно скопировать и вставить целиком. В нём уже исправлены все ошибки, включая корректную обработку массивов, переводы на русский и правильное извлечение легендарных/бонусных действий и реакций.

Вот финальная версия макроса для Foundry. Просто скопируйте его и вставьте в консоль (F12).

javascript
// Макрос Foundry: Полный экспорт NPC для трекера (исправленная версия)
(async () => {
    // Словари перевода
    const damageTranslations = {
        'necrotic': 'некротический', 'bludgeoning': 'дробящий', 'piercing': 'колющий', 'slashing': 'рубящий',
        'fire': 'огонь', 'cold': 'холод', 'lightning': 'электричество', 'acid': 'кислота', 'poison': 'яд',
        'psychic': 'психический', 'radiant': 'излучение', 'thunder': 'звук', 'force': 'силовое поле'
    };
    const conditionTranslations = {
        'charmed': 'очарованный', 'frightened': 'испуганный', 'grappled': 'схваченный', 'paralyzed': 'парализованный',
        'petrified': 'окаменевший', 'poisoned': 'отравленный', 'prone': 'сбитый с ног', 'restrained': 'опутанный',
        'stunned': 'ошеломлённый', 'unconscious': 'бессознательный', 'exhaustion': 'истощённый', 'invisible': 'невидимый',
        'blinded': 'ослеплённый', 'deafened': 'оглохший', 'incapacitated': 'недееспособный'
    };
    const languageTranslations = {
        'common': 'общий', 'abyssal': 'бездны', 'infernal': 'инфернальный', 'draconic': 'драконий',
        'elvish': 'эльфийский', 'giant': 'великаний', 'goblin': 'гоблинский', 'dwarvish': 'дварфийский',
        'orc': 'орочий', 'sylvan': 'сильван', 'primordial': 'изначальный', 'celestial': 'небесный',
        'deep speech': 'глубинная речь', 'undercommon': 'подземный', 'halfling': 'полуросликов'
    };

    // Очистка HTML/Foundry-тегов
    function cleanDescription(html) {
        if (!html) return '';
        let text = html.replace(/<[^>]+>/g, '');
        text = text.replace(/@UUID\[[^\]]+?\]\{([^}]+)\}/g, '$1');
        text = text.replace(/@Compendium\[[^\]]+?\]\{([^}]+)\}/g, '$1');
        text = text.replace(/\[\[\/r[^\]]+\]\]/g, '');
        text = text.replace(/\[\[\/damage[^\]]+\]\]/g, '');
        text = text.replace(/&[a-z]+;/g, '');
        return text.trim();
    }

    // Перевод списка (универсальная обработка массива или строки)
    function translateList(value, dict) {
        if (!value) return '';
        let arr = [];
        if (Array.isArray(value)) arr = value;
        else if (typeof value === 'string') arr = value.split(',').map(s => s.trim());
        else return '';
        return arr.map(item => dict[item.toLowerCase()] || item).join(', ');
    }

    const monsters = [];
    const actors = game.actors.filter(a => a.type === "npc");

    for (const actor of actors) {
        const sys = actor.system;
        const traits = sys.traits;
        const attr = sys.attributes;
        const details = sys.details;
        const abilities = sys.abilities;

        // --- Базовая информация ---
        const ac = attr.ac.flat ?? attr.ac.value;
        const hp = attr.hp.max;
        const cr = details.cr;
        const xp = details.xp?.value ?? (cr ? Math.floor(cr * 100) : 0);
        const size = traits.size;
        const type = details.type?.value || '';
        const alignment = details.alignment || '';

        // --- Спасброски ---
        const saves = Object.entries(abilities)
            .filter(([_, ab]) => ab.proficient)
            .map(([key, ab]) => `${key} +${ab.save}`).join(', ') || '—';

        // --- Навыки ---
        const skills = Object.entries(sys.skills)
            .filter(([_, sk]) => sk.proficient)
            .map(([key, sk]) => `${key} +${sk.total}`).join(', ') || '—';

        // --- Чувства ---
        const senses = [];
        if (attr.senses?.darkvision) senses.push(`тёмное зрение ${attr.senses.darkvision} фт.`);
        if (attr.senses?.blindsight) senses.push(`слепое зрение ${attr.senses.blindsight} фт.`);
        if (attr.senses?.tremorsense) senses.push(`чувство вибрации ${attr.senses.tremorsense} фт.`);
        if (attr.senses?.truesight) senses.push(`истинное зрение ${attr.senses.truesight} фт.`);
        if (attr.senses?.special) senses.push(attr.senses.special);
        const sensesStr = senses.join(', ') || '—';

        // --- Языки ---
        const langs = traits.languages?.value || [];
        const languagesStr = translateList(langs, languageTranslations) || '—';

        // --- Сопротивления/иммунитеты ---
        const dr = traits.dr?.value || [];
        const di = traits.di?.value || [];
        const ci = traits.ci?.value || [];
        const damage_resistances = translateList(dr, damageTranslations);
        const damage_immunities = translateList(di, damageTranslations);
        const condition_immunities = translateList(ci, conditionTranslations);

        // --- Характеристики ---
        const stats = {
            str: { score: abilities.str.value, mod: abilities.str.mod },
            dex: { score: abilities.dex.value, mod: abilities.dex.mod },
            con: { score: abilities.con.value, mod: abilities.con.mod },
            int: { score: abilities.int.value, mod: abilities.int.mod },
            wis: { score: abilities.wis.value, mod: abilities.wis.mod },
            cha: { score: abilities.cha.value, mod: abilities.cha.mod }
        };

        // --- Разбор предметов (действия, легендарки, бонусные, реакции) ---
        const items = actor.items.contents;
        const actions = [];
        const features = [];
        const legendary_actions = [];
        const bonus_actions = [];
        const reactions = [];
        const spellsList = [];

        for (const item of items) {
            const name = item.name;
            const desc = cleanDescription(item.system.description?.value || '');
            if (!desc) continue;

            const actType = item.system.activation?.type;
            const itemObj = { name, description: desc };

            if (item.type === 'weapon') {
                actions.push(itemObj);
            } else if (item.type === 'spell') {
                spellsList.push(item.id);
            } else if (item.type === 'feat') {
                if (actType === 'legendary') legendary_actions.push(itemObj);
                else if (actType === 'bonus') bonus_actions.push(itemObj);
                else if (actType === 'reaction') reactions.push(itemObj);
                else if (name.toLowerCase().includes('мультиатака') || name.toLowerCase().includes('атака')) actions.push(itemObj);
                else features.push(itemObj);
            }
        }

        // Легендарное сопротивление
        const legResMax = sys.resources?.legres?.max;
        if (legResMax) {
            features.push({
                name: `Легендарное сопротивление (${legResMax}/день)`,
                description: "Если существо проваливает спасбросок, оно может вместо этого сделать его успешным."
            });
        }

        // Описание
        const description = cleanDescription(details.biography?.value || '');

        const monster = {
            name: actor.name,
            image: actor.img,
            ac, hp, cr, xp, size, type, alignment,
            senses: sensesStr,
            languages: languagesStr,
            saves: saves,
            skills: skills,
            profBonus: attr.prof,
            stats,
            damage_resistances,
            damage_immunities,
            condition_immunities,
            actions,
            features,
            legendary_actions,
            bonus_actions,
            reactions,
            spells: spellsList,
            description
        };

        monsters.push(monster);
    }

    // Сохранение
    const jsonStr = JSON.stringify(monsters, null, 2);
    const blob = new Blob([jsonStr], { type: 'application/json' });
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = 'monsters_full_export.json';
    a.click();

    console.log(`✅ Экспортировано ${monsters.length} NPC.`);
    ui.notifications.info(`Готово! Экспортировано ${monsters.length} NPC.`);
})();

// Макрос Foundry: Полный экспорт NPC с переводами и корректными списками
(async () => {
    // Словари перевода
    const damageTranslations = {

        'fire': 'огонь', 'cold': 'холод', 'lightning': 'электричество', 'acid': 'кислота', 'poison': 'яд',
        'psychic': 'психический', 'radiant': 'излучение', 'thunder': 'звук', 'force': 'силовое поле'

    const conditionTranslations = {
        'charmed': 'очарованный', 'frightened': 'испуганный', 'grappled': 'схваченный', 'paralyzed': 'парализованный',

        'stunned': 'ошеломлённый', 'unconscious': 'бессознательный', 'exhaustion': 'истощённый', 'invisible': 'невидимый',
        'blinded': 'ослеплённый', 'deafened': 'оглохший', 'incapacitated': 'недееспособный'
    };
    const languageTranslations = {
        'common': 'общий', 'abyssal': 'бездны', 'infernal': 'инфернальный', 'draconic': 'драконий',
        'elvish': 'эльфийский', 'giant': 'великаний', 'goblin': 'гоблинский', 'dwarvish': 'дварфийский',
        'orc': 'орочий', 'sylvan': 'сильван', 'primordial': 'изначальный', 'celestial': 'небесный',
        'deep speech': 'глубинная речь', 'undercommon': 'подземный', 'halfling': 'полуросликов'

        'prf': 'Выступление', 'per': 'Убеждение', 'rel': 'Религия', 'slt': 'Ловкость рук',

    // Очистка HTML/Foundry-тегов
    function cleanDescription(html) {
        if (!html) return '';
        let text = String(html).replace(/<[^>]+>/g, '');
        text = text.replace(/@UUID\[[^\]]+?\]\{([^}]+)\}/g, '$1');
        text = text.replace(/@Compendium\[[^\]]+?\]\{([^}]+)\}/g, '$1');
        text = text.replace(/\[\[\/r[^\]]+\]\]/g, '');
        text = text.replace(/\[\[\/damage[^\]]+\]\]/g, '');
        text = text.replace(/&[a-z]+;/g, '');
        return text.trim();
    }

    // Универсальная функция перевода списка (массив, строка, CSV)
    function translateList(input, dict) {
        if (!input) return '';
        let arr = [];
        if (Array.isArray(input)) arr = input;
        else if (typeof input === 'string') arr = input.split(',').map(s => s.trim());
        else return '';
        return arr.map(item => dict[item.toLowerCase()] || item).join(', ');
    }

    const monsters = [];
    const actors = game.actors.filter(a => a.type === "npc");

    for (const actor of actors) {
        const sys = actor.system;
        const traits = sys.traits;
        const attr = sys.attributes;
        const details = sys.details;
        const abilities = sys.abilities;

        // --- Базовая информация ---
        const ac = attr.ac.flat ?? attr.ac.value;
        const hp = attr.hp.max;
        const cr = details.cr;
        const xp = details.xp?.value ?? (cr ? Math.floor(cr * 100) : 0);
        const size = traits.size;
        const type = details.type?.value || '';
        const alignment = details.alignment || '';

        // --- Спасброски (переведённые) ---
        const savesArr = Object.entries(abilities)
            .filter(([_, ab]) => ab.proficient)
            .map(([key, ab]) => `${saveTranslations[key] || key} ${ab.save >= 0 ? '+' + ab.save : ab.save}`);
        const saves = savesArr.join(', ') || '—';

        // --- Навыки (переведённые) ---
        const skillsArr = Object.entries(sys.skills)
            .filter(([_, sk]) => sk.proficient > 0)
            .map(([key, sk]) => `${skillTranslations[key] || key} ${sk.total >= 0 ? '+' + sk.total : sk.total}`);
        const skills = skillsArr.join(', ') || '—';

        // --- Чувства ---
        const sensesParts = [];

        if (attr.senses?.tremorsense) sensesParts.push(`чувство вибрации ${attr.senses.tremorsense} фт.`);
        if (attr.senses?.truesight) sensesParts.push(`истинное зрение ${attr.senses.truesight} фт.`);
        if (attr.senses?.special) sensesParts.push(attr.senses.special);
        const senses = sensesParts.join(', ') || '—';

        // --- Языки (переведённые) ---
        const languages = translateList(traits.languages?.value, languageTranslations) || '—';

        // --- Сопротивления/иммунитеты (переведённые) ---
        const damage_resistances = translateList(traits.dr?.value, damageTranslations);
        const damage_immunities = translateList(traits.di?.value, damageTranslations);
        const condition_immunities = translateList(traits.ci?.value, conditionTranslations);

        // --- Характеристики ---
        const stats = {
            str: { score: abilities.str.value, mod: abilities.str.mod },
            dex: { score: abilities.dex.value, mod: abilities.dex.mod },
            con: { score: abilities.con.value, mod: abilities.con.mod },
            int: { score: abilities.int.value, mod: abilities.int.mod },
            wis: { score: abilities.wis.value, mod: abilities.wis.mod },
            cha: { score: abilities.cha.value, mod: abilities.cha.mod }
        };

        // --- Разбор предметов ---
        const items = actor.items.contents;
        const actions = [];
        const features = [];
        const legendary_actions = [];
        const bonus_actions = [];
        const reactions = [];
        const spellsList = [];

        for (const item of items) {
            const name = item.name;
            const desc = cleanDescription(item.system.description?.value || '');
            if (!desc) continue;

            const actType = item.system.activation?.type;
            const itemObj = { name, description: desc };

            if (item.type === 'weapon') {
                actions.push(itemObj);
            } else if (item.type === 'spell') {
                spellsList.push(item.id);
            } else if (item.type === 'feat') {
                if (actType === 'legendary') legendary_actions.push(itemObj);
                else if (actType === 'bonus') bonus_actions.push(itemObj);
                else if (actType === 'reaction') reactions.push(itemObj);
                else if (name.toLowerCase().includes('мультиатака') || name.toLowerCase().includes('атака')) actions.push(itemObj);
                else features.push(itemObj);
            }
        }

        // Легендарное сопротивление
        const legResMax = sys.resources?.legres?.max;
        if (legResMax) {
            features.push({
                name: `Легендарное сопротивление (${legResMax}/день)`,
                description: "Если существо проваливает спасбросок, оно может вместо этого сделать его успешным."
            });

Я понял, в чём загвоздка: в Foundry поля languages, damage_resistances и т.д. могут храниться как массивы, строки или объекты. Плюс в старом коде translateList падал, если приходила строка. Я написал финальный, максимально надёжный макрос, который корректно обрабатывает любые форматы, переводит термины и вытягивает абсолютно всё: языки, сопротивления, иммунитеты, спасброски, навыки, чувства, легендарки, бонусные действия, реакции и описание.

Выполните этот макрос в консоли Foundry (F12). Он создаст готовый JSON, который можно сразу использовать в трекере.

javascript
// Финальный макрос Foundry: Полный экспорт NPC с переводами и всеми полями
(async () => {
    // Словари перевода
    const damageDict = {
        'necrotic':'некротический','bludgeoning':'дробящий','piercing':'колющий','slashing':'рубящий',
        'fire':'огонь','cold':'холод','lightning':'электричество','acid':'кислота','poison':'яд',
        'psychic':'психический','radiant':'излучение','thunder':'звук','force':'силовое поле'
    };
    const conditionDict = {
        'charmed':'очарованный','frightened':'испуганный','grappled':'схваченный','paralyzed':'парализованный',
        'petrified':'окаменевший','poisoned':'отравленный','prone':'сбитый с ног','restrained':'опутанный',
        'stunned':'ошеломлённый','unconscious':'бессознательный','exhaustion':'истощённый','invisible':'невидимый',
        'blinded':'ослеплённый','deafened':'оглохший','incapacitated':'недееспособный'
    };
    const langDict = {
        'common':'общий','abyssal':'бездны','infernal':'инфернальный','draconic':'драконий','elvish':'эльфийский',
        'giant':'великаний','goblin':'гоблинский','dwarvish':'дварфийский','orc':'орочий','sylvan':'сильван',
        'primordial':'изначальный','celestial':'небесный','deep speech':'глубинная речь','undercommon':'подземный',
        'halfling':'полуросликов'
    };

    // Функция приведения к массиву
    const toArray = (val) => {
        if (!val) return [];
        if (Array.isArray(val)) return val;
        if (typeof val === 'string') return val.split(',').map(s => s.trim()).filter(Boolean);
        if (typeof val === 'object' && val.value) return toArray(val.value);
        return [];
    };

    // Перевод списка
    const translateList = (list, dict) => {
        const arr = toArray(list);
        return arr.map(item => dict[item.toLowerCase()] || item).join(', ');
    };

    // Очистка описания
    const cleanDesc = (html) => {
        if (!html) return '';
        let text = html.replace(/<[^>]+>/g, '');
        text = text.replace(/@UUID\[[^\]]+?\]\{([^}]+)\}/g, '$1');
        text = text.replace(/@Compendium\[[^\]]+?\]\{([^}]+)\}/g, '$1');
        text = text.replace(/\[\[\/r[^\]]+\]\]/g, '');
        text = text.replace(/\[\[\/damage[^\]]+\]\]/g, '');
        text = text.replace(/&[a-z]+;/g, '');
        return text.trim();
    };

    const monsters = [];
    const actors = game.actors.filter(a => a.type === "npc");

    for (const actor of actors) {
        const sys = actor.system;
        const traits = sys.traits;
        const attr = sys.attributes;
        const details = sys.details;
        const abilities = sys.abilities;

        // Базовая информация
        const ac = attr.ac.flat ?? attr.ac.value;
        const hp = attr.hp.max;
        const cr = details.cr;
        const xp = details.xp?.value ?? (cr ? Math.floor(cr * 100) : 0);
        const size = traits.size;
        const type = details.type?.value || '';
        const alignment = details.alignment || '';

        // Спасброски (отображаем только proficient)
        const saves = Object.entries(abilities)
            .filter(([_, ab]) => ab.proficient)
            .map(([key, ab]) => `${key} +${ab.save}`).join(', ') || '—';

        // Навыки (proficient)
        const skills = Object.entries(sys.skills)
            .filter(([_, sk]) => sk.proficient)
            .map(([key, sk]) => `${key} +${sk.total}`).join(', ') || '—';

        // Чувства
        const sensesParts = [];
        if (attr.senses?.darkvision) sensesParts.push(`тёмное зрение ${attr.senses.darkvision} фт.`);
        if (attr.senses?.blindsight) sensesParts.push(`слепое зрение ${attr.senses.blindsight} фт.`);
        if (attr.senses?.tremorsense) sensesParts.push(`чувство вибрации ${attr.senses.tremorsense} фт.`);
        if (attr.senses?.truesight) sensesParts.push(`истинное зрение ${attr.senses.truesight} фт.`);
        if (attr.senses?.special) sensesParts.push(attr.senses.special);
        const sensesStr = sensesParts.join(', ') || '—';

        // Языки (перевод)
        const langs = toArray(traits.languages?.value);
        const languagesStr = translateList(langs, langDict) || '—';

        // Сопротивления/иммунитеты (перевод)
        const dr = toArray(traits.dr?.value);
        const di = toArray(traits.di?.value);
        const ci = toArray(traits.ci?.value);
        const damage_resistances = translateList(dr, damageDict);
        const damage_immunities = translateList(di, damageDict);
        const condition_immunities = translateList(ci, conditionDict);

        // Статы
        const stats = {
            str: { score: abilities.str.value, mod: abilities.str.mod },
            dex: { score: abilities.dex.value, mod: abilities.dex.mod },
            con: { score: abilities.con.value, mod: abilities.con.mod },
            int: { score: abilities.int.value, mod: abilities.int.mod },
            wis: { score: abilities.wis.value, mod: abilities.wis.mod },
            cha: { score: abilities.cha.value, mod: abilities.cha.mod }
        };

        // Разбор предметов
        const items = actor.items.contents;
        const actions = [];
        const features = [];
        const legendary_actions = [];
        const bonus_actions = [];
        const reactions = [];
        const spellsList = [];

        for (const item of items) {
            const name = item.name;
            const desc = cleanDesc(item.system.description?.value || '');
            if (!desc) continue;

            const actType = item.system.activation?.type;
            const itemObj = { name, description: desc };

            if (item.type === 'weapon') actions.push(itemObj);
            else if (item.type === 'spell') spellsList.push(item.id);
            else if (item.type === 'feat') {
                if (actType === 'legendary') legendary_actions.push(itemObj);
                else if (actType === 'bonus') bonus_actions.push(itemObj);
                else if (actType === 'reaction') reactions.push(itemObj);
                else if (name.toLowerCase().includes('мультиатака') || name.toLowerCase().includes('атака')) actions.push(itemObj);
                else features.push(itemObj);
            }
        }

        // Легендарное сопротивление
        const legResMax = sys.resources?.legres?.max;
        if (legResMax) {
            features.push({
                name: `Легендарное сопротивление (${legResMax}/день)`,
                description: "Если существо проваливает спасбросок, оно может вместо этого сделать его успешным."
            });
        }

        // Описание
        const description = cleanDesc(details.biography?.value || '');

        monsters.push({
            name: actor.name,
            image: actor.img,
            ac, hp, cr, xp, size, type, alignment,
            senses: sensesStr,
            languages: languagesStr,
            saves,
            skills,
            profBonus: attr.prof,
            stats,
            damage_resistances,
            damage_immunities,
            condition_immunities,
            actions,
            features,
            legendary_actions,
            bonus_actions,
            reactions,
            spells: spellsList,
            description
        });
    }

    // Сохраняем файл
    const jsonStr = JSON.stringify(monsters, null, 2);
    const blob = new Blob([jsonStr], { type: 'application/json' });
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = 'monsters_full_export.json';
    a.click();

    console.log(`✅ Экспортировано ${monsters.length} NPC.`);
    ui.notifications.info(`Готово! Экспортировано ${monsters.length} NPC.`);
})();

    function avgFromFormula(formula) {
        const match = formula?.replace(/\s/g, '').match(/(\d+)d(\d+)(?:\+(\d+))?/i);
        if (match) return Math.floor(parseInt(match[1]) * ((parseInt(match[2]) + 1) / 2) + (match[3] ? parseInt(match[3]) : 0));
        return parseInt(formula) || 30;
    }
    function rollHP(formula) {
        if (!formula) return { total: 24, rolls: [], formula };
        const match = formula.replace(/\s/g, '').match(/(\d+)d(\d+)(?:\+(\d+))?/i);
        if (!match) return { total: parseInt(formula) || 24, rolls: [], formula };
        let total = match[3] ? parseInt(match[3]) : 0;
        let rolls = [];
        for (let i = 0; i < parseInt(match[1]); i++) {
            let roll = Math.floor(Math.random() * parseInt(match[2])) + 1;
            rolls.push(roll);
            total += roll;
        }
        return { total, rolls, formula };
    }

    // ========== Очистка текста ==========
        function cleanText(str) {
        if (!str) return '';
        let cleaned = String(str);
        // Заменяем броски урона и проверки
        cleaned = cleaned.replace(/\[\[\/r\s+([^\]]+)\]\]/g, '$1');
        cleaned = cleaned.replace(/\[\[\/damage\s+([^\]]+)\]\]/g, (match, formula) => {
            let typeMatch = formula.match(/type=([a-zA-Z]+)/);
            let diceFormula = formula.replace(/\s*type=[a-zA-Z]+\s*/, '').trim();
            if (typeMatch) {
                let typeRus = translations.damage[typeMatch[1].toLowerCase()] || typeMatch[1];
                return `${diceFormula} (${typeRus})`;
            }
            return diceFormula;
        });
        // Удаляем прочие теги
		cleaned = cleaned.replace(/@Compendium\[[^\]]*?actions[^\]]*?\]\{([^}]+)\}/gi, '$1');
        cleaned = cleaned.replace(/@UUID\[[^\]]+\]/g, '');
        cleaned = cleaned.replace(/@Compendium\[[^\]]+\]/g, '');
        cleaned = cleaned.replace(/&reference\[[^\]]+\]\{([^}]+)\}/gi, '$1');
        cleaned = cleaned.replace(/&Reference\[[^\]]+\]\{([^}]+)\}/gi, '$1');
        cleaned = cleaned.replace(/&reference\[[^\]]+\]/g, '');
        cleaned = cleaned.replace(/&Reference\[[^\]]+\]/g, '');
        cleaned = cleaned.replace(/reference\[[^\]]+\]/gi, '');
        cleaned = cleaned.replace(/&[a-zA-Z]+;/g, '');
        cleaned = cleaned.replace(/\{@[^}]+\}/g, '');
        cleaned = cleaned.replace(/\(\)/g, '');
        cleaned = cleaned.replace(/\(\s*\)/g, '');
        cleaned = cleaned.replace(/\(\s*\+\d+\s*\)/g, '');
        cleaned = cleaned.replace(/[\[\]\(\)]/g, ' ');
        // Убираем мусор Foundry
        cleaned = cleaned.replace(/\[\[lookup\s+@name(?:\s+capitalize)?\]\]/gi, '');
        cleaned = cleaned.replace(/lookup\s+@name(?:\s+capitalize)?/gi, '');
        cleaned = cleaned.replace(/^Примечание\s*[—–-]\s*/i, '');
        cleaned = cleaned.replace(/\s+/g, ' ').trim();
        return cleaned || '—';
    }
    function extractString(value) {
        if (!value) return '';
        if (typeof value === 'string') return cleanText(value);
        if (Array.isArray(value)) return value.map(v => extractString(v)).join(', ');
        if (value instanceof Set) return Array.from(value).map(v => extractString(v)).join(', ');
        if (typeof value === 'object') {
            if (value.value) {
                if (typeof value.value === 'string') return cleanText(value.value);
                if (value.value.data) {
                    if (typeof value.value.data === 'string') return cleanText(value.value.data);
                    if (value.value.data.content) return extractProseMirrorContent(value.value.data.content);
                }
            }
            if (value.data) {
                if (typeof value.data === 'string') return cleanText(value.data);
                if (value.data.content) return extractProseMirrorContent(value.data.content);
            }
        }
        return '';
    }
    function extractProseMirrorContent(content) {
        if (!Array.isArray(content)) return '';
        let text = '';
        for (let node of content) {
            if (node.type === 'paragraph' || node.type === 'heading') {
                if (node.content) text += extractProseMirrorContent(node.content) + '\n';
            } else if (node.type === 'text') {
                text += node.text || '';
            } else if (node.content) {
                text += extractProseMirrorContent(node.content);
            }
        }
        return cleanText(text);
    }

    // ========== Поиск заклинания по русскому названию ==========
    function findSpellByName(rusName) {
        const lower = rusName.toLowerCase();
        return allSpellsList.find(s => {
            const fullName = s.spellName.toLowerCase();
            const rusPart = fullName.split('/')[0].trim();
            return rusPart === lower || fullName === lower;
        });
    }

                   function extractSpellsFromDescription(desc) {
    if (!desc) return [];
    const foundSpells = new Set();
    
    // Нормализация: удаляем всё после слеша, убираем лишние пробелы
    function normalizeName(name) {
        return name.split('/')[0].trim().toLowerCase();
    }
    
    // --- 1. Foundry-теги с названием ---
    const compendiumRegex = /@Compendium\[[^\]]*?spells[^\]]*?\]\{([^}]+)\}/gi;
    let match;
    while ((match = compendiumRegex.exec(desc)) !== null) {
        const norm = normalizeName(match[1]);
        const spell = allSpellsList.find(s => normalizeName(s.spellName) === norm);
        if (spell) foundSpells.add(spell.spellName);
    }
    
    const uuidWithNameRegex = /@UUID\[[^\]]*?\]\{([^}]+)\}/gi;
    while ((match = uuidWithNameRegex.exec(desc)) !== null) {
        const norm = normalizeName(match[1]);
        const spell = allSpellsList.find(s => normalizeName(s.spellName) === norm);
        if (spell) foundSpells.add(spell.spellName);
    }
    
    // --- 2. @UUID без скобок ---
    const uuidNoBraceRegex = /@UUID\[[^\]]*?\]\s*([^,\.\{]+?)(?=\s*(?:,|\.|\{|$))/gi;
    while ((match = uuidNoBraceRegex.exec(desc)) !== null) {
        let name = match[1].trim().replace(/\s*\([^)]*\)/g, '').trim();
        if (name) {
            const norm = normalizeName(name);
            let spell = allSpellsList.find(s => normalizeName(s.spellName) === norm);
            if (!spell) {
                const nextWords = match[0].split(/\s+/).slice(1, 3).join(' ');
                const norm2 = normalizeName(nextWords);
                spell = allSpellsList.find(s => normalizeName(s.spellName) === norm2);
            }
            if (spell) foundSpells.add(spell.spellName);
        }
    }
    
    // --- 3. Поиск по ID (UUID без названия)
const uuidIdRegex = /@UUID\[[^\]]*?\.([^\]\.]+)\]/gi;
while ((match = uuidIdRegex.exec(desc)) !== null) {
    const id = match[1].trim();
    console.log(`[DEBUG] Ищем заклинание по ID: "${id}"`);
    
    // Ищем по полю _id (или id) в allSpellsList
    let spell = allSpellsList.find(s => s._id === id || s.id === id);
    if (!spell) {
        // Пробуем найти по совпадению в конце ID
        spell = allSpellsList.find(s => (s._id || '').endsWith(id) || (s.id || '').endsWith(id));
        if (spell) console.log(`[DEBUG] Найдено по окончанию ID: ${spell.spellName}`);
    }
    if (!spell) {
        // Преобразуем camelCase в слова и ищем по ним
        const nameFromId = id.replace(/([a-z])([A-Z])/g, '$1 $2').toLowerCase();
        console.log(`[DEBUG] Преобразованный ID в слова: "${nameFromId}"`);
        spell = allSpellsList.find(s => s.spellName.toLowerCase().includes(nameFromId));
        if (spell) console.log(`[DEBUG] Найдено по имени: ${spell.spellName}`);
    }
    if (spell) {
        foundSpells.add(spell.spellName);
        console.log(`[DEBUG] Успешно добавлено: ${spell.spellName}`);
    } else {
        console.warn(`[DEBUG] Заклинание с ID "${id}" не найдено в allSpellsList`);
    }
}
    
    // --- 4. Обычные списки ---
    const listPattern = /(?:Неограниченно|(?:\d+\/день(?:\s+каждое)?)|Заговоры)\s*[:;]?\s*([^\.]+?)(?=\s*(?:\.|$|\d\/|Неограниченно|Описание|$))/gi;
    const cleanedDesc = cleanText(desc);
    let listMatch;
    while ((listMatch = listPattern.exec(cleanedDesc)) !== null) {
        const segment = listMatch[1].trim();
        const parts = segment.split(/[\s,;]+/);
        for (let i = 0; i < parts.length; i++) {
            const part = parts[i];
            if (part.length < 3) continue;
            const norm = normalizeName(part);
            let spell = allSpellsList.find(s => normalizeName(s.spellName) === norm);
            if (!spell && i + 1 < parts.length) {
                const twoWords = part + ' ' + parts[i+1];
                const norm2 = normalizeName(twoWords);
                spell = allSpellsList.find(s => normalizeName(s.spellName) === norm2);
                if (spell) i++;
            }
            if (!spell && i + 2 < parts.length) {
                const threeWords = part + ' ' + parts[i+1] + ' ' + parts[i+2];
                const norm3 = normalizeName(threeWords);
                spell = allSpellsList.find(s => normalizeName(s.spellName) === norm3);
                if (spell) i += 2;
            }
            if (spell) foundSpells.add(spell.spellName);
        }
    }
    
    return Array.from(foundSpells);
}

    // ========== Подсветка заклинаний в тексте ==========
    function highlightSpellsInText(text) {
    if (!text) return text;
    
    // 1. Обрабатываем Foundry-ссылки
    let result = text;
    result = result.replace(/@Compendium\[[^\]]*?spells[^\]]*?\]\{([^}]+)\}/gi, (match, spellName) => {
        const found = findSpellByName(spellName.trim());
        if (found) return `<span class="spell-link" data-spell="${found.spellName}">${spellName.trim()}</span>`;
        return match;
    });
    result = result.replace(/@UUID\[[^\]]*?\]\{([^}]+)\}/gi, (match, spellName) => {
        const found = findSpellByName(spellName.trim());
        if (found) return `<span class="spell-link" data-spell="${found.spellName}">${spellName.trim()}</span>`;
        return match;
    });
    
    // 2. Очищаем текст
    let cleaned = cleanText(result);
    
    // 3. Сортируем заклинания от длинных к коротким
    const sortedSpells = [...allSpellsList].sort((a, b) => {
        const aName = a.spellName.split('/')[0].trim();
        const bName = b.spellName.split('/')[0].trim();
        return bName.length - aName.length;
    });
    
    // 4. Проходим по каждому заклинанию
    for (let spell of sortedSpells) {
        const rusName = spell.spellName.split('/')[0].trim();
        if (rusName.length < 3) continue;
        
        // Экранируем спецсимволы (кроме слеша и дефиса, которые могут быть в названии)
        const escaped = rusName.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
        // Границы: пробел, начало строки, теги, знаки препинания
        const regex = new RegExp('(^|\\s|>|\\b|[;:])(' + escaped + ')(?=\\s|,|;|\\.|$|<|\\b)', 'gi');
        
        cleaned = cleaned.replace(regex, (match, p1, p2) => {
            // Проверяем, не внутри ли мы уже span'а
            const pos = cleaned.indexOf(match);
            const before = cleaned.substring(0, pos);
            if (before.lastIndexOf('<span class="spell-link"') > before.lastIndexOf('</span>')) {
                return match;
            }
            return p1 + `<span class="spell-link" data-spell="${spell.spellName}">${p2}</span>`;
        });
    }
    
    return cleaned;
}

    // ========== Парсинг ячеек ==========
           function parseSpellSlots(description) {
        const slots = {};
        // Поддержка "2-й круг", "2 круг", "2-го круга", "2 круга" и т.д.
        const regex = /(\d+)(?:-й|-го)?\s*круг[а]?\s*\((\d+)\s*ячейк[а-я]*\)/gi;
        let match;
        while ((match = regex.exec(description)) !== null) {
            const level = parseInt(match[1]);
            const count = parseInt(match[2]);
            if (!isNaN(level) && !isNaN(count)) slots[level] = { max: count, used: 0 };
        }
        return slots;
    }

    let dragSrcId = null;
    function attachDragEvents(card, id) {
        card.setAttribute('draggable', 'true');
        card.ondragstart = (e) => { dragSrcId = id; e.dataTransfer.effectAllowed = 'move'; card.classList.add('dragging'); };
        card.ondragend = (e) => { card.classList.remove('dragging'); dragSrcId = null; };
        card.ondragover = (e) => e.preventDefault();
        card.ondrop = (e) => {
            e.preventDefault();
            if (dragSrcId === null || dragSrcId === id) return;
            const srcIndex = combatants.findIndex(c => c.id == dragSrcId);
            const destIndex = combatants.findIndex(c => c.id == id);
            if (srcIndex !== -1 && destIndex !== -1) {
                [combatants[srcIndex], combatants[destIndex]] = [combatants[destIndex], combatants[srcIndex]];
                if (currentTurnIndex === srcIndex) currentTurnIndex = destIndex;
                else if (currentTurnIndex === destIndex) currentTurnIndex = srcIndex;
                renderCombat(); scrollToCurrentTurn(); saveCurrentStateToSession();
            }
            dragSrcId = null;
        };
    }

        let spellAbility = "cha"; // по умолчанию Харизма
        if (data.spellsInfo && data.spellsInfo.base) {
            if (data.spellsInfo.base.code) spellAbility = data.spellsInfo.base.code.toLowerCase();
            else if (data.spellsInfo.base.value) spellAbility = data.spellsInfo.base.value.toLowerCase();
        }
        
        // Получаем модификатор этой характеристики
        let spellMod = 0;
        const stats = data.stats || {};
        if (stats[spellAbility] && stats[spellAbility].score) {
            spellMod = Math.floor((stats[spellAbility].score - 10) / 2);
        }
        
        const spellSaveDC = 8 + profBonus + spellMod;
        const spellAttack = profBonus + spellMod;
        
        // Собираем статы для карточки
        const heroStats = { 
            str: { score: stats.str?.score || 10, mod: Math.floor((stats.str?.score - 10)/2) || 0 }, 
            dex: { score: stats.dex?.score || 10, mod: Math.floor((stats.dex?.score - 10)/2) || 0 }, 
            con: { score: stats.con?.score || 10, mod: Math.floor((stats.con?.score - 10)/2) || 0 }, 
            int: { score: stats.int?.score || 10, mod: Math.floor((stats.int?.score - 10)/2) || 0 }, 
            wis: { score: stats.wis?.score || 10, mod: Math.floor((stats.wis?.score - 10)/2) || 0 }, 
            cha: { score: stats.cha?.score || 10, mod: Math.floor((stats.cha?.score - 10)/2) || 0 } 
        };
        
        const newHero = { 
            id: nextId++, 
            name, 
            race, 
            class: charClass, 
            ac, 
            maxHp, 
            currentHp, 
            tempHp: 0, 
            hpFormula: "5d8+10", 
            initBonus: dexMod, 
            initiative: 0, 
            conditions: new Set(), 
            exhaustion: 0, 
            isMonster: false, 
            log: [], 
            avatar, 
            spellSaveDC, 
            spellAttack, 
            stats: heroStats, 
            heroData: data, 
            spellSlots: {} 
        };
        
        combatants.push(newHero);
        addLog(newHero, `📜 Импорт персонажа ${name}`);
        renderCombat(); 
        saveCurrentStateToSession();
    } catch(e) { 
        alert("Ошибка разбора JSON: " + e.message); 
    }
}

    function getConditionClass(cond) {
        const map = {
            "Бессознательный": "cond-unconscious", "Испуганный": "cond-frightened", "Истощённый": "cond-exhaustion",
            "Невидимый": "cond-invisible", "Недееспособный": "cond-incapacitated", "Оглохший": "cond-deafened",
            "Окаменевший": "cond-petrified", "Опутанный": "cond-restrained", "Ослеплённый": "cond-blinded",
            "Отравленный": "cond-poisoned", "Очарованный": "cond-charmed", "Ошеломлённый": "cond-stunned",
            "Парализованный": "cond-paralyzed", "Сбитый с ног": "cond-prone", "Схваченный": "cond-grappled",
            "Концентрация": "cond-concentration"
        };
        return map[cond] || "";
    }
