    // ========== Очистка текста ==========
        function cleanText(str) {
        if (!str) return '';
        let cleaned = String(str);
        // Заменяем броски урона и проверки
        cleaned = cleaned.replace(/\[\[\/r\s+([^\]]+)\]\]/g, '$1');
        cleaned = cleaned.replace(/\[\[\/damage\s+([^\]]+)\]\]/g, (match, formula) => {
            let typeMatch = formula.match(/type=([a-zA-Z]+)/);
            let diceFormula = formula.replace(/\s*type=[a-zA-Z]+\s*/, '').trim();
            if (typeMatch) {
                let typeRus = translations.damage[typeMatch[1].toLowerCase()] || typeMatch[1];
                return `${diceFormula} (${typeRus})`;
            }
            return diceFormula;
        });
        // Удаляем прочие теги
        cleaned = cleaned.replace(/@UUID\[[^\]]+\]/g, '');
        cleaned = cleaned.replace(/@Compendium\[[^\]]+\]/g, '');
        cleaned = cleaned.replace(/&reference\[[^\]]+\]\{([^}]+)\}/gi, '$1');
        cleaned = cleaned.replace(/&Reference\[[^\]]+\]\{([^}]+)\}/gi, '$1');
        cleaned = cleaned.replace(/&reference\[[^\]]+\]/g, '');
        cleaned = cleaned.replace(/&Reference\[[^\]]+\]/g, '');
        cleaned = cleaned.replace(/reference\[[^\]]+\]/gi, '');
        cleaned = cleaned.replace(/&[a-zA-Z]+;/g, '');
        cleaned = cleaned.replace(/\{@[^}]+\}/g, '');
        cleaned = cleaned.replace(/\(\)/g, '');
        cleaned = cleaned.replace(/\(\s*\)/g, '');
        cleaned = cleaned.replace(/\(\s*\+\d+\s*\)/g, '');
        cleaned = cleaned.replace(/[\[\]\(\)]/g, ' ');
        // Убираем мусор Foundry
        cleaned = cleaned.replace(/\[\[lookup\s+@name(?:\s+capitalize)?\]\]/gi, '');
        cleaned = cleaned.replace(/lookup\s+@name(?:\s+capitalize)?/gi, '');
        cleaned = cleaned.replace(/^Примечание\s*[—–-]\s*/i, '');
        cleaned = cleaned.replace(/\s+/g, ' ').trim();
        return cleaned || '—';
    }
    function extractString(value) {
        if (!value) return '';
        if (typeof value === 'string') return cleanText(value);
        if (Array.isArray(value)) return value.map(v => extractString(v)).join(', ');
        if (value instanceof Set) return Array.from(value).map(v => extractString(v)).join(', ');
        if (typeof value === 'object') {
            if (value.value) {
                if (typeof value.value === 'string') return cleanText(value.value);
                if (value.value.data) {
                    if (typeof value.value.data === 'string') return cleanText(value.value.data);
                    if (value.value.data.content) return extractProseMirrorContent(value.value.data.content);
                }
            }
            if (value.data) {
                if (typeof value.data === 'string') return cleanText(value.data);
                if (value.data.content) return extractProseMirrorContent(value.data.content);
            }
        }
        return '';
    }
    function extractProseMirrorContent(content) {
        if (!Array.isArray(content)) return '';
        let text = '';
        for (let node of content) {
            if (node.type === 'paragraph' || node.type === 'heading') {
                if (node.content) text += extractProseMirrorContent(node.content) + '\n';
            } else if (node.type === 'text') {
                text += node.text || '';
            } else if (node.content) {
                text += extractProseMirrorContent(node.content);
            }
        }
        return cleanText(text);
    }

    // ========== Поиск заклинания по русскому названию ==========
    function findSpellByName(rusName) {
        const lower = rusName.toLowerCase();
        return allSpellsList.find(s => {
            const fullName = s.spellName.toLowerCase();
            const rusPart = fullName.split('/')[0].trim();
            return rusPart === lower || fullName === lower;
        });
    }

                   function extractSpellsFromDescription(desc) {
    if (!desc) return [];
    const foundSpells = new Set();
    
    // Нормализация: удаляем всё после слеша, убираем лишние пробелы
    function normalizeName(name) {
        return name.split('/')[0].trim().toLowerCase();
    }
    
    // --- 1. Foundry-теги с названием ---
    const compendiumRegex = /@Compendium\[[^\]]*?spells[^\]]*?\]\{([^}]+)\}/gi;
    let match;
    while ((match = compendiumRegex.exec(desc)) !== null) {
        const norm = normalizeName(match[1]);
        const spell = allSpellsList.find(s => normalizeName(s.spellName) === norm);
        if (spell) foundSpells.add(spell.spellName);
    }
    
    const uuidWithNameRegex = /@UUID\[[^\]]*?\]\{([^}]+)\}/gi;
    while ((match = uuidWithNameRegex.exec(desc)) !== null) {
        const norm = normalizeName(match[1]);
        const spell = allSpellsList.find(s => normalizeName(s.spellName) === norm);
        if (spell) foundSpells.add(spell.spellName);
    }
    
    // --- 2. @UUID без скобок ---
    const uuidNoBraceRegex = /@UUID\[[^\]]*?\]\s*([^,\.\{]+?)(?=\s*(?:,|\.|\{|$))/gi;
    while ((match = uuidNoBraceRegex.exec(desc)) !== null) {
        let name = match[1].trim().replace(/\s*\([^)]*\)/g, '').trim();
        if (name) {
            const norm = normalizeName(name);
            let spell = allSpellsList.find(s => normalizeName(s.spellName) === norm);
            if (!spell) {
                const nextWords = match[0].split(/\s+/).slice(1, 3).join(' ');
                const norm2 = normalizeName(nextWords);
                spell = allSpellsList.find(s => normalizeName(s.spellName) === norm2);
            }
            if (spell) foundSpells.add(spell.spellName);
        }
    }
    
    // --- 3. Поиск по ID (UUID без названия)
const uuidIdRegex = /@UUID\[[^\]]*?\.([^\]\.]+)\]/gi;
while ((match = uuidIdRegex.exec(desc)) !== null) {
    const id = match[1].trim();
    console.log(`[DEBUG] Ищем заклинание по ID: "${id}"`);
    
    // Ищем по полю _id (или id) в allSpellsList
    let spell = allSpellsList.find(s => s._id === id || s.id === id);
    if (!spell) {
        // Пробуем найти по совпадению в конце ID
        spell = allSpellsList.find(s => (s._id || '').endsWith(id) || (s.id || '').endsWith(id));
        if (spell) console.log(`[DEBUG] Найдено по окончанию ID: ${spell.spellName}`);
    }
    if (!spell) {
        // Преобразуем camelCase в слова и ищем по ним
        const nameFromId = id.replace(/([a-z])([A-Z])/g, '$1 $2').toLowerCase();
        console.log(`[DEBUG] Преобразованный ID в слова: "${nameFromId}"`);
        spell = allSpellsList.find(s => s.spellName.toLowerCase().includes(nameFromId));
        if (spell) console.log(`[DEBUG] Найдено по имени: ${spell.spellName}`);
    }
    if (spell) {
        foundSpells.add(spell.spellName);
        console.log(`[DEBUG] Успешно добавлено: ${spell.spellName}`);
    } else {
        console.warn(`[DEBUG] Заклинание с ID "${id}" не найдено в allSpellsList`);
    }
}
    
    // --- 4. Обычные списки ---
    const listPattern = /(?:Неограниченно|(?:\d+\/день(?:\s+каждое)?)|Заговоры)\s*[:;]?\s*([^\.]+?)(?=\s*(?:\.|$|\d\/|Неограниченно|Описание|$))/gi;
    const cleanedDesc = cleanText(desc);
    let listMatch;
    while ((listMatch = listPattern.exec(cleanedDesc)) !== null) {
        const segment = listMatch[1].trim();
        const parts = segment.split(/[\s,;]+/);
        for (let i = 0; i < parts.length; i++) {
            const part = parts[i];
            if (part.length < 3) continue;
            const norm = normalizeName(part);
            let spell = allSpellsList.find(s => normalizeName(s.spellName) === norm);
            if (!spell && i + 1 < parts.length) {
                const twoWords = part + ' ' + parts[i+1];
                const norm2 = normalizeName(twoWords);
                spell = allSpellsList.find(s => normalizeName(s.spellName) === norm2);
                if (spell) i++;
            }
            if (!spell && i + 2 < parts.length) {
                const threeWords = part + ' ' + parts[i+1] + ' ' + parts[i+2];
                const norm3 = normalizeName(threeWords);
                spell = allSpellsList.find(s => normalizeName(s.spellName) === norm3);
                if (spell) i += 2;
            }
            if (spell) foundSpells.add(spell.spellName);
        }
    }
    
    return Array.from(foundSpells);
}

    // ========== Подсветка заклинаний в тексте ==========
    function highlightSpellsInText(text) {
    if (!text) return text;
    
    // 1. Обрабатываем Foundry-ссылки
    let result = text;
    result = result.replace(/@Compendium\[[^\]]*?spells[^\]]*?\]\{([^}]+)\}/gi, (match, spellName) => {
        const found = findSpellByName(spellName.trim());
        if (found) return `<span class="spell-link" data-spell="${found.spellName}">${spellName.trim()}</span>`;
        return match;
    });
    result = result.replace(/@UUID\[[^\]]*?\]\{([^}]+)\}/gi, (match, spellName) => {
        const found = findSpellByName(spellName.trim());
        if (found) return `<span class="spell-link" data-spell="${found.spellName}">${spellName.trim()}</span>`;
        return match;
    });
    
    // 2. Очищаем текст
    let cleaned = cleanText(result);
    
    // 3. Сортируем заклинания от длинных к коротким
    const sortedSpells = [...allSpellsList].sort((a, b) => {
        const aName = a.spellName.split('/')[0].trim();
        const bName = b.spellName.split('/')[0].trim();
        return bName.length - aName.length;
    });
    
    // 4. Проходим по каждому заклинанию
    for (let spell of sortedSpells) {
        const rusName = spell.spellName.split('/')[0].trim();
        if (rusName.length < 3) continue;
        
        // Экранируем спецсимволы (кроме слеша и дефиса, которые могут быть в названии)
        const escaped = rusName.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
        // Границы: пробел, начало строки, теги, знаки препинания
        const regex = new RegExp('(^|\\s|>|\\b|[;:])(' + escaped + ')(?=\\s|,|;|\\.|$|<|\\b)', 'gi');
        
        cleaned = cleaned.replace(regex, (match, p1, p2) => {
            // Проверяем, не внутри ли мы уже span'а
            const pos = cleaned.indexOf(match);
            const before = cleaned.substring(0, pos);
            if (before.lastIndexOf('<span class="spell-link"') > before.lastIndexOf('</span>')) {
                return match;
            }
            return p1 + `<span class="spell-link" data-spell="${spell.spellName}">${p2}</span>`;
        });
    }
    
    return cleaned;
}

    // ========== Парсинг ячеек ==========
           function parseSpellSlots(description) {
        const slots = {};
        // Поддержка "2-й круг", "2 круг", "2-го круга", "2 круга" и т.д.
        const regex = /(\d+)(?:-й|-го)?\s*круг[а]?\s*\((\d+)\s*ячейк[а-я]*\)/gi;
        let match;
        while ((match = regex.exec(description)) !== null) {
            const level = parseInt(match[1]);
            const count = parseInt(match[2]);
            if (!isNaN(level) && !isNaN(count)) slots[level] = { max: count, used: 0 };
        }
        return slots;
    }

    function applyDamage(c, dmg) { let remaining = dmg; if (c.tempHp > 0) { let absorb = Math.min(c.tempHp, remaining); c.tempHp -= absorb; remaining -= absorb; addLog(c, `🛡️ Врем.ХП поглотило ${absorb}`); } if (remaining > 0) { c.currentHp = Math.max(0, c.currentHp - remaining); addLog(c, `⚔️ Урон ${dmg} → ❤️ ${c.currentHp}/${c.maxHp}`); } renderCombat(); saveCurrentStateToSession(); }
    function applyHeal(c, amount) { c.currentHp = Math.min(c.maxHp, c.currentHp + amount); addLog(c, `💚 Лечение +${amount} → ${c.currentHp}/${c.maxHp}`); renderCombat(); saveCurrentStateToSession(); }

    let dragSrcId = null;
    function attachDragEvents(card, id) {
        card.setAttribute('draggable', 'true');
        card.ondragstart = (e) => { dragSrcId = id; e.dataTransfer.effectAllowed = 'move'; card.classList.add('dragging'); };
        card.ondragend = (e) => { card.classList.remove('dragging'); dragSrcId = null; };
        card.ondragover = (e) => e.preventDefault();
        card.ondrop = (e) => {
            e.preventDefault();
            if (dragSrcId === null || dragSrcId === id) return;
            const srcIndex = combatants.findIndex(c => c.id == dragSrcId);
            const destIndex = combatants.findIndex(c => c.id == id);
            if (srcIndex !== -1 && destIndex !== -1) {
                [combatants[srcIndex], combatants[destIndex]] = [combatants[destIndex], combatants[srcIndex]];
                if (currentTurnIndex === srcIndex) currentTurnIndex = destIndex;
                else if (currentTurnIndex === destIndex) currentTurnIndex = srcIndex;
                renderCombat(); scrollToCurrentTurn(); saveCurrentStateToSession();
            }
            dragSrcId = null;
        };
    }

📜 Макрос для Foundry: создание полного UUID-словаря
javascript
// Макрос: Собрать все UUID из NPC и создать словарь ID -> название
(async () => {
    const uuidMap = { spells: {}, actions: {}, items: {}, other: {} };

    // Получить все NPC
    const npcs = game.actors.filter(a => a.type === "npc");
    
    // Регулярка для поиска @UUID[.ID] или @UUID[Compendium....ID]
    const uuidRegex = /@UUID\[([^\]]+)\]/g;
    
    for (const npc of npcs) {
        // Обходим features и actions
        const items = [...(npc.items.contents || [])];
        for (const item of items) {
            const desc = item.system?.description?.value || '';
            let match;
            while ((match = uuidRegex.exec(desc)) !== null) {
                const fullUuid = match[1];
                // Извлекаем ID (последняя часть после точки)
                const parts = fullUuid.split('.');
                const id = parts[parts.length - 1];
                
                // Пытаемся найти документ по UUID
                let doc = null;
                try {
                    doc = await fromUuid(fullUuid);
                } catch (e) {}
                
                if (doc) {
                    const type = doc.type;
                    const name = doc.name;
                    if (type === 'spell') uuidMap.spells[id] = name;
                    else if (type === 'action' || type === 'feat') uuidMap.actions[id] = name;
                    else if (type === 'item') uuidMap.items[id] = name;
                    else uuidMap.other[id] = name;
                } else {
                    // Если документ не найден, оставляем пустым или пытаемся извлечь имя из текста
                    if (!uuidMap.spells[id] && !uuidMap.actions[id]) {
                        // Можно попробовать найти в компендиумах
                        const fromPack = await findInCompendiums(fullUuid);
                        if (fromPack) {
                            if (fromPack.type === 'spell') uuidMap.spells[id] = fromPack.name;
                            else if (fromPack.type === 'action') uuidMap.actions[id] = fromPack.name;
                        }
                    }
                }
            }
        }
    }
    
    // Функция поиска в компендиумах
    async function findInCompendiums(uuid) {
        // Пытаемся загрузить документ напрямую из компендиума
        const doc = await fromUuid(uuid);
        if (doc) return { type: doc.type, name: doc.name };
        
        // Если не вышло, пробуем искать по ID в известных компендиумах
        const packs = ['laaru-dnd5-hw.spells', 'laaru-dnd5-hw.actions', 'dnd5e.spells', 'dnd5e.items'];
        const id = uuid.split('.').pop();
        for (const packName of packs) {
            const pack = game.packs.get(packName);
            if (!pack) continue;
            const index = await pack.getIndex();
            const entry = index.find(e => e._id === id);
            if (entry) return { type: entry.type || 'unknown', name: entry.name };
        }
        return null;
    }

    // Сохраняем файл
    const jsonStr = JSON.stringify(uuidMap, null, 2);
    const handle = await window.showSaveFilePicker({
        suggestedName: 'uuid-map-complete.json',
        types: [{ description: 'JSON', accept: { 'application/json': ['.json'] } }]
    });
    const writable = await handle.createWritable();
    await writable.write(jsonStr);
    await writable.close();
    
    ui.notifications.info(`UUID-словарь сохранён. Заклинаний: ${Object.keys(uuidMap.spells).length}, действий: ${Object.keys(uuidMap.actions).length}`);
})();

Понял, это действительно самый надёжный путь. Ваш предыдущий макрос был нацелен на актёров, а нужные данные «спрятаны» в библиотеках-компендиумах (Compendiums), на которые ссылаются эти @UUID. Да, их можно вытащить напрямую оттуда.

Вот план, как это сделать, с кодом, который можно сразу скопировать.

🎯 План действий
Найти нужные компендиумы: Сначала в консоли Foundry (F12) узнаем точные ID компендиумов, которые содержат заклинания (spells) и действия (actions).

Написать макрос: Этот макрос пройдётся по всем документам внутри этих компендиумов и соберёт их внутренние ID (_id) и названия (name), создав полную карту соответствий.

Сохранить результат: Макрос сформирует и сохранит готовый JSON-файл (uuid-map.json), который вы потом подключите к своему трекеру.

🛠️ Шаг 1: Находим точные ID компендиумов
Откройте в Foundry консоль (F12) и выполните эту команду:

javascript
console.log(Array.from(game.packs.keys()).filter(key => key.includes('spell') || key.includes('action') || key.includes('laaru')))
Эта команда покажет вам полные ID всех активных компендиумов, в названии которых есть spell, action или laaru. Пример того, что вы можете увидеть:

'laaru-dnd5-hw.spells'

'laaru-dnd5-hw.actions'

Скопируйте эти ID.

🪄 Шаг 2: Запускаем макрос для сбора данных
Создайте новый макрос в Foundry, вставьте в него код ниже и выполните.

В коде обязательно замените PACK_IDS на те ID, которые вы нашли в первом шаге.

javascript
// Список ID компендиумов, из которых нужно собрать данные (замените на свои)
const PACK_IDS = [
    'laaru-dnd5-hw.spells',
    'laaru-dnd5-hw.actions'
];

const uuidMap = { spells: {}, actions: {}, items: {}, other: {} };

async function buildUuidMap() {
    for (const packId of PACK_IDS) {
        const pack = game.packs.get(packId);
        if (!pack) {
            console.warn(`Компендиум с ID "${packId}" не найден.`);
            continue;
        }

        // Определяем тип документа, чтобы положить в нужный раздел
        let category = 'other';
        if (pack.metadata.type === 'Item') {
            // Пытаемся угадать категорию по названию компендиума
            if (packId.includes('spell')) category = 'spells';
            else if (packId.includes('action')) category = 'actions';
            else category = 'items';
        }

        console.log(`Обработка компендиума: ${packId} (категория: ${category})`);

        // Получаем все документы из компендиума
        const documents = await pack.getDocuments();
        for (const doc of documents) {
            if (doc.id && doc.name) {
                uuidMap[category][doc.id] = doc.name;
            }
        }
    }

    // Формируем и сохраняем JSON-файл
    const jsonStr = JSON.stringify(uuidMap, null, 2);
    const blob = new Blob([jsonStr], { type: 'application/json' });
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = 'uuid-map.json';
    a.click();

    console.log('UUID-словарь успешно сохранён!');
}

buildUuidMap();

Научите функцию resolveUuid пользоваться этим словарём. Найдите в коде функцию resolveUuid (если её нет, создайте) и замените её на эту:

javascript
function resolveUuid(uuid) {
    // Извлекаем ID (последний сегмент после точки)
    const match = uuid.match(/\.([^\.\]\[]+)(?:\]|$)/);
    if (!match) return null;
    const id = match[1];
    
    // Ищем во всех категориях загруженного uuidMap
    for (const category in uuidMap) {
        if (uuidMap[category][id]) {
            return uuidMap[category][id];
        }
    }
    return null;
}

🪄 Макрос для сбора UUID-словаря
javascript
// Макрос для Foundry VTT: Сбор всех заклинаний, действий и предметов из указанных компендиумов
(async () => {
    // Список ID компендиумов, которые нужно обработать (из вашего вывода)
    const PACK_IDS = [
        'laaru-dnd5-hw.spells',
        'laaru-dnd5-hw.actions',
        'laaru-dnd5-hw.items',    // на всякий случай
        'dnd5e.spells',           // SRD заклинания
        'dnd5e.spells24'          // 2024 заклинания
    ];

    const uuidMap = { spells: {}, actions: {}, items: {}, other: {} };

    for (const packId of PACK_IDS) {
        const pack = game.packs.get(packId);
        if (!pack) {
            console.warn(`Компендиум "${packId}" не найден.`);
            continue;
        }

        // Определяем категорию по названию
        let category = 'other';
        if (packId.includes('spell')) category = 'spells';
        else if (packId.includes('action')) category = 'actions';
        else if (packId.includes('item')) category = 'items';

        console.log(`Обработка компендиума: ${packId} (категория: ${category})`);

        // Получаем документы — важно: сначала загружаем содержимое компендиума
        const documents = await pack.getDocuments();
        console.log(`Найдено ${documents.length} документов`);

        for (const doc of documents) {
            // Используем doc.id (или doc._id) — уникальный идентификатор
            const id = doc.id || doc._id;
            if (id && doc.name) {
                uuidMap[category][id] = doc.name;
            }
        }
    }

    // Сохраняем файл
    const jsonStr = JSON.stringify(uuidMap, null, 2);
    const blob = new Blob([jsonStr], { type: 'application/json' });
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = 'uuid-map.json';
    a.click();

    console.log('UUID-словарь успешно сохранён!');
    ui.notifications.info('UUID-словарь сохранён. Проверьте загрузки.');
})();

🔍 Почему раньше могло не работать
Компендиум мог быть не загружен в память. Метод getDocuments() сам загружает его при необходимости.

Возможно, вы использовали pack.index вместо pack.getDocuments(). В индексе могут быть не все поля, а только _id и name, но для полного словаря это тоже подходит. Однако getDocuments() даёт полные документы с гарантией.

Я подготовил универсальный макрос для Foundry, который обойдёт вообще все компендиумы, которые есть в вашем мире, и создаст единый словарь uuid-map.json.

🧙 Универсальный макрос (сбор всех данных из всех компендиумов)
javascript
// Макрос Foundry: Собрать ВСЕ ID из ВСЕХ компендиумов
(async () => {
    const fullMap = {};

    // Получаем все компендиумы, доступные в мире
    const allPacks = game.packs.contents;
    console.log(`Найдено компендиумов: ${allPacks.length}`);

    for (const pack of allPacks) {
        const packId = pack.metadata.id || pack.collection;
        console.log(`Обработка: ${packId} (${pack.metadata.type})`);

        try {
            // Загружаем индекс (быстро) или полные документы
            await pack.getIndex();
            const index = pack.index;
            if (!index) continue;

            // Если в карте ещё нет раздела для этого типа, создадим
            const type = pack.metadata.type || 'other';
            if (!fullMap[type]) fullMap[type] = {};

            for (const entry of index) {
                // entry._id — это и есть тот самый ID, который в UUID после точки
                if (entry._id && entry.name) {
                    fullMap[type][entry._id] = entry.name;
                }
            }
        } catch (e) {
            console.warn(`Не удалось обработать компендиум ${packId}:`, e);
        }
    }

    // Сохраняем результат
    const jsonStr = JSON.stringify(fullMap, null, 2);
    const blob = new Blob([jsonStr], { type: 'application/json' });
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = 'uuid-map-full.json';
    a.click();

    console.log('✅ Полный UUID-словарь сохранён!');
    ui.notifications.info('Словарь сохранён. Проверьте загрузки.');
})();

🔧 Как использовать в трекере
После того как вы получили uuid-map-full.json, интегрируйте его в код трекера так же, как мы планировали ранее, но с одной поправкой: искать нужно по всем категориям.

Функция resolveUuid для полного словаря:

javascript
let uuidMapFull = {};

async function loadUuidMap() {
    try {
        const res = await fetch('uuid-map-full.json');
        if (res.ok) uuidMapFull = await res.json();
        console.log('Полный UUID-словарь загружен');
    } catch(e) {
        console.warn('Ошибка загрузки uuid-map-full.json');
    }
}

function resolveUuid(uuid) {
    const match = uuid.match(/\.([^\.\]\[]+)(?:\]|$)/);
    if (!match) return null;
    const id = match[1];
    for (const category in uuidMapFull) {
        if (uuidMapFull[category][id]) {
            return uuidMapFull[category][id];
        }
    }
    return null;
}

const ids = [
    "lOXVls1gG8hhZ2Au", "iPEHRT9XLFFbg2ft", "mBxtg70UvNrT51VZ", "M3R9sLjY4stB4Msy",
    "5cmiXYRKQdAwWLCG", "exrpv2OuCw1D0ovB", "qc7Lbo9aZuXa4VOR", "pAyQP6jSr5RC2K35",
    "hbru4aUBTBkZEgrd", "3oeG1FaoJB7w9LFF", "Hrb2WcvQQoSuebbh", "oMCqZ5dUYlWvgSGd",
    "bXuiY0kc0JgZv66L"

(async () => {
